175 vec_dbl_Type solution_c;
176 vec_dbl_Type solution_d;
177 for (UN T=0; T<assemblyFEElements_.size(); T++) {

180 solution_c = getSolution(elementsSolid->getElement(T).getVectorNodeList(), c_rep,1);
