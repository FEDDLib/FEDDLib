#
# A) Define the project
#

# To be safe, define your minimum CMake version.  This may be newer than the
# min required by TriBITS.

cmake_minimum_required(VERSION 3.23.0 FATAL_ERROR)

include(${CMAKE_SOURCE_DIR}/ProjectName.cmake)
project(${PROJECT_NAME} NONE)

# The version number.
set(FEDDLIB_VERSION_MAJOR 1)
set(FEDDLIB_VERSION_MINOR 0)
set(FEDDLIB_VERSION_PATCH 0)
set(${PROJECT_NAME}_VERSION "${FEDDLIB_VERSION_MAJOR}.${FEDDLIB_VERSION_MINOR}.${FEDDLIB_VERSION_PATCH}")

set(${PROJECT_NAME}_ENABLE_Fortran ON CACHE BOOL "Enable Fortran compiler")

# Enable CMake config file generation and installation
# This allows consumers to use find_package(FEDDLib) and get proper imported targets
set(${PROJECT_NAME}_ENABLE_INSTALL_CMAKE_CONFIG_FILES ON CACHE BOOL
    "Enable installation of CMake config files for find_package()")

# Generate export file dependencies to ensure transitive dependencies are properly
# included in the exported targets (e.g., Fortran runtime libraries)
set(${PROJECT_NAME}_GENERATE_EXPORT_FILE_DEPENDENCIES ON CACHE BOOL
    "Generate export file dependencies for proper transitive linking")

# Set C++ standard to C++17
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

#
# B) Pull in the TriBITS system and execute
#

include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/tribits/TriBITS.cmake)

# CMake requires this be in the top file and not in an include file :-(
cmake_minimum_required(VERSION ${TRIBITS_CMAKE_MINIMUM_REQUIRED})

# Add source dir to includes
tribits_include_directories(${CMAKE_SOURCE_DIR})

# Do all of the processing for this Tribits project
tribits_project()


foreach(ENABLED_PACKAGE ${FEDDLib_ENABLED_PACKAGES})
	string(TOUPPER ${ENABLED_PACKAGE} UPPER_PACKAGE)
	set(FEDD_HAVE_${UPPER_PACKAGE} ON)
endforeach()

configure_file(${CMAKE_SOURCE_DIR}/FEDDLib_Config.h.in
               ${CMAKE_BINARY_DIR}/FEDDLib_Config.h)

# Install the generated configuration header
install(FILES ${CMAKE_BINARY_DIR}/FEDDLib_Config.h DESTINATION include)

# Install git_version.h (generated by feddlib/problems/abstract/CMakeLists.txt)
# This file is included by Problem_decl.hpp
if(EXISTS ${CMAKE_BINARY_DIR}/git_version.h)
    install(FILES ${CMAKE_BINARY_DIR}/git_version.h DESTINATION include)
endif()
